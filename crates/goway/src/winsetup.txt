    # 1. install and start the OpenSSH server
    Add-WindowsCapability -Online -Name OpenSSH.Server~~~~0.0.1.0
    Set-Service -Name sshd -StartupType Automatic; Start-Service sshd
    # 2. authorize goway's key (administrator accounts use administrators_authorized_keys)
    $f = "$env:ProgramData\ssh\administrators_authorized_keys"
    Add-Content -Force -Path $f -Value '{KEY}'
    icacls.exe $f /inheritance:r /grant "Administrators:F" /grant "SYSTEM:F"
    # (a non-administrator account uses $env:USERPROFILE\.ssh\authorized_keys instead)
    # 3. make PowerShell the login shell
    New-ItemProperty -Path HKLM:\SOFTWARE\OpenSSH -Name DefaultShell -PropertyType String -Force -Value C:\Windows\System32\WindowsPowerShell\v1.0\powershell.exe
    Restart-Service sshd
